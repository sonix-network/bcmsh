Cint recipes

Small Cint programs that read state through the SDK API on a Trident 3 SONiC switch. All of them are read-only: they call only *_get and *_traverse functions and change nothing on the switch. Each starts with cint_reset(); so it can be run again, and ends by printing DONE <name>; if that line is missing, the program did not run to the end. See Cint: the C interpreter for how Cint runs and The Cint language for the language.

Copy a program to /root in the syncd container and run it from the host (see Cint: the C interpreter for why not /tmp):

Hardware-validated

$ sudo docker cp hello_info.c syncd:/root/
$ bcmcmd -t 120 "cint /root/hello_info.c"

or from an interactive bcmsh session with cint /root/hello_info.c.

Programs that walk a table hold that table’s lock while they run, so syncd waits to update it. They print only the first entries, but they still visit every entry; on a switch with a large L2 or route table, run them when a short delay in programming is acceptable.

Chip identity

bcm_info_get returns the PCI vendor and device ID and the revision. The Trident 3 below reports device 0xb870; other members of the family report other IDs, for example 0xb873 in the capture in the cint reference. The structure printed by print info ends with per-die arrays that are all zero on Trident 3.

hello_info.c
/*
 * hello_info.c - print the chip identity with bcm_info_get().
 * Read-only.  Run:  cint /root/hello_info.c
 */
cint_reset();

int hello_info()
{
    int unit = 0;
    int rv;
    bcm_info_t info;

    rv = bcm_info_get(unit, &info);
    if (rv != BCM_E_NONE) {
        printf("bcm_info_get failed: %d (%s)\n", rv, bcm_errmsg(rv));
        return rv;
    }
    printf("vendor 0x%04x device 0x%04x revision 0x%02x\n",
           info.vendor, info.device, info.revision);
    print info;
    return 0;
}

hello_info();
printf("DONE hello_info\n");

Port status

One line per front-panel port with the admin state, link state, speed and untagged VLAN. Port numbers are SDK logical port numbers, as shown by ps; see Ports and port bitmaps.

port_status.c
/*
 * port_status.c - one line per front-panel port: enable, link, speed and
 * untagged VLAN.  Read-only.  Run:  cint /root/port_status.c
 */
cint_reset();

int ps_show(int unit, bcm_port_t port)
{
    int enable = -1;
    int link = -1;
    int speed = -1;
    bcm_vlan_t pvid = 0;
    int vid;
    int rv;

    rv = bcm_port_enable_get(unit, port, &enable);
    if (rv != BCM_E_NONE) {
        printf("port %3d: bcm_port_enable_get: %s\n", port, bcm_errmsg(rv));
        return rv;
    }
    bcm_port_link_status_get(unit, port, &link);
    bcm_port_speed_get(unit, port, &speed);
    bcm_port_untagged_vlan_get(unit, port, &pvid);
    vid = pvid;
    printf("port %3d: enable %d link %d speed %6d pvid %4d\n",
           port, enable, link, speed, vid);
    return 0;
}

int ps_main()
{
    int unit = 0;
    int rv;
    int count = 0;
    bcm_port_t port;
    bcm_pbmp_t pbm;
    bcm_port_config_t config;

    bcm_port_config_t_init(&config);
    rv = bcm_port_config_get(unit, &config);
    if (rv != BCM_E_NONE) {
        printf("bcm_port_config_get failed: %s\n", bcm_errmsg(rv));
        return rv;
    }
    pbm = config.port;
    printf("front-panel ports: ${pbm}\n");
    BCM_PBMP_ITER(pbm, port) {
        ps_show(unit, port);
        count++;
    }
    printf("%d ports\n", count);
    return 0;
}

ps_main();
printf("DONE port_status\n");

Port counters

Reads 64-bit counters of one port with bcm_stat_get. The port number is the first argument; without one, port 1 is read. These are the SDK’s counters, the same ones show counters reads, so they are not reset by sonic-clear counters.

port_counters.c
/*
 * port_counters.c - read a few SNMP-style counters of one port with
 * bcm_stat_get().  Read-only.
 * Run:  cint /root/port_counters.c [<logical port>]     (default: port 1)
 */
cint_reset();

int pc_atoi(char *s)
{
    int i = 0;
    int v = 0;
    int c;

    while (1) {
        c = s[i];
        if (c < '0' || c > '9') {
            break;
        }
        v = v * 10 + (c - '0');
        i++;
    }
    return v;
}

int pc_show(int unit, bcm_port_t port, bcm_stat_val_t stat, char *name)
{
    uint64 val;
    int rv;

    COMPILER_64_ZERO(&val);
    rv = bcm_stat_get(unit, port, stat, &val);
    if (rv != BCM_E_NONE) {
        printf("  %-22s error %d (%s)\n", name, rv, bcm_errmsg(rv));
        return rv;
    }
    printf("  %-22s ${val}  hi %u lo %u\n", name,
           COMPILER_64_HI(&val), COMPILER_64_LO(&val));
    return 0;
}

int pc_main()
{
    int unit = 0;
    bcm_port_t port = 1;

    if (ARGC > 0) {
        port = pc_atoi(ARGV[0]);
    }
    printf("port %d\n", port);
    pc_show(unit, port, snmpIfHCInOctets, "snmpIfHCInOctets");
    pc_show(unit, port, snmpIfHCInUcastPkts, "snmpIfHCInUcastPkts");
    pc_show(unit, port, snmpIfInErrors, "snmpIfInErrors");
    pc_show(unit, port, snmpIfInDiscards, "snmpIfInDiscards");
    pc_show(unit, port, snmpIfHCOutOctets, "snmpIfHCOutOctets");
    pc_show(unit, port, snmpIfHCOutUcastPkts, "snmpIfHCOutUcastPkts");
    pc_show(unit, port, snmpIfOutErrors, "snmpIfOutErrors");
    pc_show(unit, port, snmpIfOutDiscards, "snmpIfOutDiscards");
    return 0;
}

pc_main();
printf("DONE port_counters\n");

L3 table usage

How full the route, host, next-hop and interface tables are, from bcm_l3_info. l3 info shows similar numbers.

l3_info.c
/*
 * l3_info.c - L3 table sizes and usage from bcm_l3_info().
 * Read-only.  Run:  cint /root/l3_info.c
 */
cint_reset();

int li_main()
{
    int unit = 0;
    int rv;
    bcm_l3_info_t info;

    rv = bcm_l3_info(unit, &info);
    if (rv != BCM_E_NONE) {
        printf("bcm_l3_info failed: %s\n", bcm_errmsg(rv));
        return rv;
    }
    printf("routes  used %d of %d\n", info.l3info_used_route, info.l3info_max_route);
    printf("hosts   used %d of %d\n", info.l3info_used_host, info.l3info_max_host);
    printf("egress  used %d of %d\n", info.l3info_used_nexthop, info.l3info_max_nexthop);
    printf("intf    used %d of %d\n", info.l3info_used_intf, info.l3info_max_intf);
    print info;
    return 0;
}

li_main();

Routes

Walks the IPv4 and then the IPv6 route table with bcm_l3_route_traverse and prints the first 20 of each. The third and fourth arguments of bcm_l3_route_traverse are the first and last position in the walk, counted from 0, not table indices; a route with several ECMP paths may count once per path. The intf column is the egress object or ECMP group, as in l3 defip show. IPv6 addresses print as eight full groups (fe80:0000:0000:…), without :: shortening.

l3_routes.c
/*
 * l3_routes.c - print the first routes of the IPv4 and IPv6 route tables
 * with bcm_l3_route_traverse() and a Cint callback.  Read-only.
 * Run:  cint /root/l3_routes.c
 */
cint_reset();

int lr_seen = 0;

int lr_v4_cb(int unit, int index, bcm_l3_route_t *info, void *user_data)
{
    bcm_ip_t net = info->l3a_subnet;
    bcm_ip_t mask = info->l3a_ip_mask;
    int vrf = info->l3a_vrf;
    int intf = info->l3a_intf;
    int flags = info->l3a_flags;

    lr_seen++;
    printf("v4 %3d vrf %d ${net} mask ${mask} intf %d flags 0x%x\n",
           index, vrf, intf, flags);
    return BCM_E_NONE;
}

int lr_v6_cb(int unit, int index, bcm_l3_route_t *info, void *user_data)
{
    bcm_ip6_t net = info->l3a_ip6_net;
    bcm_ip6_t mask = info->l3a_ip6_mask;
    int vrf = info->l3a_vrf;
    int intf = info->l3a_intf;
    int flags = info->l3a_flags;

    lr_seen++;
    printf("v6 %3d vrf %d ${net} mask ${mask} intf %d flags 0x%x\n",
           index, vrf, intf, flags);
    return BCM_E_NONE;
}

int lr_main()
{
    int unit = 0;
    int rv;
    int limit = 20;

    lr_seen = 0;
    rv = bcm_l3_route_traverse(unit, 0, 0, limit - 1, lr_v4_cb, NULL);
    printf("IPv4: rv %d (%s), %d shown (limit %d)\n", rv, bcm_errmsg(rv), lr_seen, limit);

    lr_seen = 0;
    rv = bcm_l3_route_traverse(unit, BCM_L3_IP6, 0, limit - 1, lr_v6_cb, NULL);
    printf("IPv6: rv %d (%s), %d shown (limit %d)\n", rv, bcm_errmsg(rv), lr_seen, limit);
    return 0;
}

lr_main();
printf("DONE l3_routes\n");

Egress objects

Lists L3 egress objects (next hops) with their MAC address, VLAN and port. The numbers match the INTF column of l3 egress show.

l3_egress.c
/*
 * l3_egress.c - list L3 egress objects (next hops) with
 * bcm_l3_egress_traverse().  Prints the first 50.  Read-only.
 * Run:  cint /root/l3_egress.c
 */
cint_reset();

int le_count = 0;
int le_limit = 50;

int le_cb(int unit, bcm_if_t intf, bcm_l3_egress_t *info, void *user_data)
{
    int vlan = info->vlan;

    le_count++;
    if (le_count > le_limit) {
        return BCM_E_NONE;
    }
    printf("egress %d: mac %02x:%02x:%02x:%02x:%02x:%02x vlan %d port %d module %d trunk %d intf %d flags 0x%x\n",
           intf,
           info->mac_addr[0], info->mac_addr[1], info->mac_addr[2],
           info->mac_addr[3], info->mac_addr[4], info->mac_addr[5],
           vlan, info->port, info->module, info->trunk, info->intf,
           info->flags);
    return BCM_E_NONE;
}

int le_main()
{
    int unit = 0;
    int rv;

    le_count = 0;
    rv = bcm_l3_egress_traverse(unit, le_cb, NULL);
    printf("rv %d (%s), %d egress objects\n", rv, bcm_errmsg(rv), le_count);
    return 0;
}

le_main();
printf("DONE l3_egress\n");

L2 table

Counts the MAC table and prints the first 30 entries. Compare l2 show.

l2_table.c
/*
 * l2_table.c - count the L2 (MAC) table and print the first 30 entries
 * with bcm_l2_traverse().  Read-only.  Run:  cint /root/l2_table.c
 */
cint_reset();

int lt2_count = 0;
int lt2_static = 0;
int lt2_limit = 30;

int lt2_cb(int unit, bcm_l2_addr_t *info, void *user_data)
{
    int vid = info->vid;

    lt2_count++;
    if (info->flags & BCM_L2_STATIC) {
        lt2_static++;
    }
    if (lt2_count > lt2_limit) {
        return BCM_E_NONE;
    }
    printf("%02x:%02x:%02x:%02x:%02x:%02x vlan %4d port %3d modid %d tgid %d flags 0x%x\n",
           info->mac[0], info->mac[1], info->mac[2],
           info->mac[3], info->mac[4], info->mac[5],
           vid, info->port, info->modid, info->tgid, info->flags);
    return BCM_E_NONE;
}

int lt2_main()
{
    int unit = 0;
    int rv;

    lt2_count = 0;
    lt2_static = 0;
    rv = bcm_l2_traverse(unit, lt2_cb, NULL);
    printf("rv %d (%s), %d entries (%d static)\n", rv, bcm_errmsg(rv), lt2_count, lt2_static);
    return 0;
}

lt2_main();
printf("DONE l2_table\n");

VLANs

Lists every VLAN with its member and untagged ports. bcm_vlan_list allocates the list; the program frees it with bcm_vlan_list_destroy.

vlans.c
/*
 * vlans.c - list VLANs with their member and untagged ports using
 * bcm_vlan_list().  Read-only; frees the list it receives.
 * Run:  cint /root/vlans.c
 */
cint_reset();

int vl_main()
{
    int unit = 0;
    int rv;
    int i;
    int count = 0;
    int vid;
    bcm_vlan_t dflt;
    bcm_vlan_data_t *list = NULL;
    bcm_pbmp_t pbm;
    bcm_pbmp_t ubm;

    rv = bcm_vlan_default_get(unit, &dflt);
    vid = dflt;
    printf("default VLAN %d (rv %d)\n", vid, rv);

    rv = bcm_vlan_list(unit, &list, &count);
    if (rv != BCM_E_NONE) {
        printf("bcm_vlan_list failed: %s\n", bcm_errmsg(rv));
        return rv;
    }
    for (i = 0; i < count; i++) {
        vid = list[i].vlan_tag;
        pbm = list[i].port_bitmap;
        ubm = list[i].ut_port_bitmap;
        printf("vlan %4d ports ${pbm} untagged ${ubm}\n", vid);
    }
    printf("%d VLANs\n", count);
    bcm_vlan_list_destroy(unit, list, count);
    return 0;
}

vl_main();
printf("DONE vlans\n");

KNET interfaces

Lists the KNET network interfaces, the Linux network devices through which SONiC sends and receives packets on each port, with the SDK port each is bound to. See KNET and the CPU port.

knet_netifs.c
/*
 * knet_netifs.c - list KNET network interfaces (the kernel netdevs that
 * SONiC uses for its EthernetN ports) and the SDK port each is bound to.
 * Read-only.  Run:  cint /root/knet_netifs.c
 */

Field-processor groups

Lists the field-processor groups (SONiC ACL tables and the SAI’s own trap rules are implemented as field-processor groups), how many entries each has, and the priority of up to 16 entries per group. fp show prints the full contents.

field_groups.c
/*
 * field_groups.c - list field-processor (ACL) groups, their usage and the
 * priority of their first entries.  Read-only.
 * Run:  cint /root/field_groups.c
 */
cint_reset();

int fg_groups = 0;

int fg_cb(int unit, bcm_field_group_t group, void *user_data)
{
    bcm_field_group_status_t st;
    bcm_field_entry_t entries[16];
    int n = 0;
    int i;
    int prio;
    int rv;

    fg_groups++;
    rv = bcm_field_group_status_get(unit, group, &st);
    if (rv != BCM_E_NONE) {
        printf("group %d: status error %s\n", group, bcm_errmsg(rv));
        return BCM_E_NONE;
    }
    printf("group %d: %d entries, %d free of %d, prio %d..%d, %d counters\n",
           group, st.entry_count, st.entries_free, st.entries_total,
           st.prio_min, st.prio_max, st.counter_count);
    rv = bcm_field_entry_multi_get(unit, group, 16, entries, &n);
    for (i = 0; rv == BCM_E_NONE && i < n; i++) {
        prio = 0;
        bcm_field_entry_prio_get(unit, entries[i], &prio);
        printf("    entry %d prio %d\n", entries[i], prio);
    }
    return BCM_E_NONE;
}

int fg_main()
{
    int unit = 0;
    int rv;

    fg_groups = 0;
    rv = bcm_field_group_traverse(unit, fg_cb, NULL);
    printf("rv %d (%s), %d groups\n", rv, bcm_errmsg(rv), fg_groups);
    return 0;
}

fg_main();
printf("DONE field_groups\n");

/*
 * enum_values.c - resolve SDK enum names to numbers in this build and read
 * one switch control.  Read-only.  Run:  cint /root/enum_values.c
 */

int ev_main()
{
    int unit = 0;
    int rv;
    int value;
    int number;
    bcm_switch_control_t ctl;
    bcm_stat_val_t stat;

    ctl = bcmSwitchL3EgressMode;
    number = ctl;
    printf("bcmSwitchL3EgressMode = %d\n", number);
    print ctl;

    stat = snmpIfHCInOctets;
