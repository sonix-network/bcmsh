L2

Note

This command belongs to the shell mode for newer (logical-table) switch chips and is not available on Trident 3.

Manage L2 (MAC) addresses

Built-in help

Usages:
      l2 add [Module=<mod>] [Port=<port>] [MACaddress=<mac>] [Vlanid=<id>]
             [Trunk=true|false] [TrunkGroupId=<id>]
             [STatic=true|false] [SourceCosPriority=true|false]
             [DiscardSource=true|false] [DiscardDest]=true|false]
             [HIT=true|false] [CPUmirror=true|false] [Group=<group>]
             [Egress_if=<egress interface>] [PortDbId=port_db_id]
            - Add L2 address associated with port
      l2 del [MACaddress=<mac>] [Vlanid=<id>] [PortDbId=port_db_id]
            - Delete L2 address with given mac and vlanid
      l2 get [MACaddress=<mac>] [Vlanid=<id>] [PortDbId=port_db_id]
            - Look up L2 address with given mac and vlanid
      l2 show
            - Show all L2 addresses
      l2 count
            - Show current L2 address count
      l2 clear [Module=<n>] [Port=<port>] [MACaddress=<mac>] [PortDbId=port_db_id]
               [Vlanid=<id>] [TrunkGroupID=<id>] [Static=true|false]
            - Remove all L2 entries on the given module, module/port,
               VLAN, or trunk group ID
      l2 replace [Module=<n>] [Port=<port>] [MACaddress=<mac>]
                 [Vlanid=<id>] [Trunk=true|false] [TrunkGroupId=<id>]
                 [STatic=true|false] [NewModule=<n>] [NewPort=<port>]
                 [NewTrunkGroupId=<id>] [PortDbId=port_db_id]
            - Replace all L2 entries matching given module/port or mac
              vlanid, trunk with new destination
      l2 age [Timer=<n>|on|off]
            - Setup L2 age timer in seconds
              If not specify timer value, then display setting
            - On|Off
              Off is to stop L2 age without breaking current interval
              On is to resume L2 age with current interval
      l2 station add [Priority=<val>] [ID=<Sid>]
             [MACaddress=<mac>] [MACaddressMask=<mask>]
             [Vlanid=<id>] [VlanidMask=<mask>]
             [Ivid=<id>] [IvidMask=<mask>]
             [Vfi=<id>] [VfiMask=<mask>]
             [SourcePort=<id>] [SourcePortMask=<mask>]
             [IPv4=true|false] [IPv6=true|false] [ArpRarp=true|false]
             [MPLS=true|false] [MiM=true|false] [TRILL=true|false]
             [FCoE=true|false] [OAM=true|false] [CPUmirror=true|false]
             [Replace=true|false] [Discard=true|false]
            - Add L2 station entry
      l2 station delete ID=<sid>
            - Delete L2 station entry
      l2 station clear
            - Delete all L2 station entries
      l2 station show [ID=<sid>]
            - Show L2 station entry, if not specify sid, then display all
      l2 station size
            - Show L2 station size
      l2 init
            - Init L2 module
      l2 detach
            - Detach L2 module
      l2 learn [on|off]
            - Enable/Disable global L2 learning
              On is to enable, off is to disable
              If no specify on or off, display current L2 learning setting
      l2 cache add [CacheIndex=<index>]
             [MACaddress=<mac>] [MACaddressMask=<mac>]
             [Vlanid=<id>] [VlanidMask=<mask>] [LookupClass=<class>]
             [SourcePort=<id>] [SourcePortMask=<mask>]
             [Module=<modid>] [Port=<port>]
             [Trunk=true|false] [TrunkGroupId=<id>]
             [SetPriority=true|false] [PRIOrity=<prio>]
             [DiscardDest]=true|false] [CPUmirror=true|false]
             [BPDU=true|false] [L3=true|false]
             [ReplacePriority=true|false] [LearnDisable=true|false]
             [PROTocolPacKeT=true|false]
            - Add L2 cache entry
      l2 cache del CacheIndex=<index> [Count=<value>]
            - Delete L2 cache entry
      l2 cache show
            - Show L2 cache entries
      l2 cache clear [ReInit=true|false]
            - Delete all L2 cache entries
      l2 cache size
            - Show L2 cache table size
      l2 size
            - Show L2 table size
      l2 dense add [MACaddress=<mac>] [Vlanid=<id>]
             [SouRCe=true|false] [DeSTination=true|false]
             [OUICOMPressed=true|false]
             [Replace=true|false]
             [Port=<port>] [Trunk=true|false] [TrunkGroupId=<id>]
             [SetPriority=true|false] [PRIOrity=<prio>]
             [STatic=true|false] [SourceCosPriority=true|false]
             [DiscardSource=true|false]
             [HIT=true|false] [Group=<group>]
            - Add an L2 dense address entry. Set Replace to update if entry exists
      l2 dense get [MACaddress=<mac>] [Vlanid=<id>]
             [SouRCe=true|false] [DeSTination=true|false]
             [OUICOMPressed=true|false]
            - Get an L2 dense address entry
      l2 dense delete [MACaddress=<mac>] [Vlanid=<id>]
             [SouRCe=true|false] [DeSTination=true|false]
             [OUICOMPressed=true|false]
            - Delete an L2 dense address entry
      l2 dense show [SouRCe=true|false] [DeSTination=true|false]
             [OUICOMPressed=true|false]
            - Show all L2 dense address entries of specified type
      l2 dense clear
            - Delete all L2 dense address entries
      l2 dense perf OPs=<val> [loops=<val>]
             [nsets=<val>] [UseApi=<bool>]
           - Test L2 dense address operation performance.
           - OPs: B0 - Add, B1 - Update, B2 - Lookup, B3 - Delete.
           - loops: Number of runs. Forced to 1 if no delete operation.
           - nsets: Number of sets.
           - UseApi: 1 - Use BCM API.
      l2 OUICOMPression add [Priority=<val>]
             [MACaddress=<mac>] [MACaddressMask=<mask>]
             [COMPressedOUI=<compressedoui>]
            - Add a MAC OUI compression entry
      l2 OUICOMPression replace [Priority=<val>]
             [MACaddress=<mac>] [MACaddressMask=<mask>]
             [COMPressedOUI=<compressedoui>]
            - Replace an existing MAC OUI compression entry or add a new one
      l2 OUICOMPression get
             [MACaddress=<mac>] [MACaddressMask=<mask>]
            - Get a MAC OUI compression entry with given MAC OUI and mask
      l2 OUICOMPression delete
             [MACaddress=<mac>] [MACaddressMask=<mask>]
            - Delete a MAC OUI compression entry
      l2 OUICOMPression clear
            - Delete all MAC OUI compression entries
